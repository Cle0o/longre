        function getDefaultSettings() {
            return FIXED_KEYS.map(key => ({

                label: DEFAULT_LABELS[key] || key,
                fixed: true
            }));
        }

        function getSettings() {
            try {
                const raw = localStorage.getItem(SETTINGS_KEY);
                if (!raw) return getDefaultSettings();
                const parsed = JSON.parse(raw);
                const defaults = getDefaultSettings();
                const existingKeys = parsed.map(c => c.key);
                defaults.forEach(def => {
                    if (!existingKeys.includes(def.key)) {
                        parsed.push(def);
                    }
                });
                parsed.forEach(c => {
                    if (FIXED_KEYS.includes(c.key)) {
                        c.fixed = true;
                    }
                });
                return parsed;
            } catch {
                return getDefaultSettings();
            }
        }

        function saveSettingsToStorage(settings) {
            localStorage.setItem(SETTINGS_KEY, JSON.stringify(settings));
        }

        function calcStatus(student) {

        function render() {

            const search = document.getElementById('searchInput').value.trim().toLowerCase();
            const yearFilter = document.getElementById('yearFilter').value;
            const statusFilter = document.getElementById('statusFilter').value;

            let filtered = students.filter(s => {
                if (search) {
                    const haystack = (s.name + s.undergraduate + s.targetSchool + s.appliedSchools + s.admittedSchools)
                        .toLowerCase();
                    if (!haystack.includes(search)) return false;
                }
                if (yearFilter && s.applyYear !== yearFilter) return false;
                if (statusFilter) {
                    const st = calcStatus(s);
                    if (st !== statusFilter) return false;
                }
                return true;
            });

            filtered.sort((a, b) => (b.createdAt || '').localeCompare(a.createdAt || ''));

            updateStats(students);
            updateYearFilter(students);
            renderTableHeader();
            renderTableBody(filtered, students.length);
        }

        function saveStudent() {
            const name = document.getElementById('f_name').value.trim();
            if (!name) {
                showToast('请输入学生姓名', 'error');
                document.getElementById('f_name').focus();
                return;
            }

            const editId = document.getElementById('editId').value;
            const students = getStudents();
            const settings = getSettings();

            const data = {};
            settings.forEach(col => {
                const key = col.key;
                const el = document.getElementById('f_' + key);
                if (el) {
                    data[key] = el.value.trim();
                }
            });
            const extraKeys = ['internship', 'research', 'competition', 'cv', 'rl', 'ps', 'transcript',
                'studyCert', 'scoringRule', 'visaProcess', 'cvProgress', 'rl1Progress', 'rl2Progress',
                'gExamScore', 'psStatus', 'gExamNote'
            ];
            extraKeys.forEach(k => {
                const el = document.getElementById('f_' + k);
                if (el) {
                    data[k] = el.value.trim();
                }
            });
            ['cvProgress', 'rl1Progress', 'rl2Progress'].forEach(k => {
                data[k] = parseInt(data[k]) || 0;
                data[k] = Math.min(100, Math.max(0, data[k]));
            });

            data.name = name;

            if (editId) {
                const idx = students.findIndex(s => s.id === editId);
                if (idx === -1) { showToast('数据异常，请刷新', 'error'); return; }
                students[idx] = { ...students[idx], ...data };
                saveStudents(students);
                showToast('✅ 学生信息已更新', 'success');
            } else {
                const newStudent = {
                    id: genId(),
                    ...data,
                    createdAt: new Date().toISOString(),
                };
                students.push(newStudent);
                saveStudents(students);
                showToast('🎉 学生添加成功！', 'success');
            }

            closeFormModal();
            render();
        }

                { key: 'rl1Progress', label: 'RL1 进度 (%)' },
                { key: 'rl2Progress', label: 'RL2 进度 (%)' },
                { key: 'gExamScore', label: 'G类 分数' },
                { key: 'psStatus', label: 'PS 状态' },
                { key: 'gExamNote', label: 'G类 备注' },
            ];
            const allFields = [...fieldMap, ...extraDetailFields];

            let html = `<div class="detail-grid">`;
            allFields.forEach(({ key, label }) => {
                const val = s[key] !== undefined ? s[key] : '';
                const isEmpty = (val === '' || val === null || val === undefined);
                const displayVal = isEmpty ? '未填写' : val;
                const full = ['account', 'targetSchool', 'scoringRule', 'internship', 'research', 'competition',
                    'visaProcess', 'schoolRequirement', 'visitRecord'
                ].includes(key) || key.startsWith('col_');
                html += `
                        <div class="detail-item ${full ? 'full' : ''}">
                            <span class="label">${escHtml(label)}</span>
                            <span class="value ${isEmpty ? 'empty' : ''}">${escHtml(String(displayVal)).replace(/\n/g, '<br>')}</span>
                        </div>
                    `;
            });
            html += `</div>`;
            body.innerHTML = html;
            document.getElementById('detailModal').classList.add('active');
        }

        // ================================================================
        //  删除、导入导出
        // ================================================================
        function deleteStudent(id) {
            if (!confirm('确定要删除该学生的所有数据吗？此操作不可撤销！')) return;
            let students = getStudents();
            const filtered = students.filter(s => s.id !== id);
            if (filtered.length === students.length) {
                showToast('学生不存在', 'error');
                return;
            }
            saveStudents(filtered);
            showToast('🗑️ 已删除该学生', 'success');
            render();
        }

        function exportData() {
            const students = getStudents();
            if (students.length === 0) {
                showToast('没有数据可导出', 'error');
                return;
            }
            const json = JSON.stringify(students, null, 2);
            const blob = new Blob([json], { type: 'application/json' });
            const url = URL.createObjectURL(blob);
            const a = document.createElement('a');
            a.href = url;
            a.download = `留学学生数据_${new Date().toISOString().slice(0,10)}.json`;
            a.click();
            URL.revokeObjectURL(url);
            showToast('📤 数据导出成功', 'success');
        }

        function importData(event) {
            const file = event.target.files[0];
            if (!file) return;
            const reader = new FileReader();
            reader.onload = (e) => {
                try {
                    const data = JSON.parse(e.target.result);
                    if (!Array.isArray(data) || data.length === 0) {
                        showToast('数据格式无效或为空', 'error');
                        return;
                    }
                    if (!data[0].name && !data[0].id) {
                        showToast('数据格式不匹配，请检查文件', 'error');
                        return;
                    }
                    if (!confirm(`将导入 ${data.length} 条学生数据，是否继续？\n（现有数据将被覆盖）`)) return;
                    saveStudents(data);
                    showToast(`📥 成功导入 ${data.length} 条数据`, 'success');
                    render();
                } catch (err) {
                    showToast('导入失败：' + err.message, 'error');
                }
            };
            reader.readAsText(file);
            event.target.value = '';
        }

        function openSettings() {
            renderSettingsList();
            document.getElementById('settingsModal').classList.add('active');
        }

        function closeSettings() {
            document.getElementById('settingsModal').classList.remove('active');
        }

        function renderSettingsList() {
            const settings = getSettings();
            const container = document.getElementById('settingsList');
            let html = '';
            settings.forEach((col) => {
                const isFixed = col.fixed || false;
                const lockIcon = isFixed ? '🔒 ' : '';
                html += `
                        <div class="settings-column-item">
                            <span class="col-label">${lockIcon}${escHtml(col.label)}</span>
                            <input type="text" id="setting_label_${col.key}" value="${escHtml(col.label)}" />
                            <div class="col-actions">
                                ${!isFixed ? `<button class="btn btn-danger btn-xs" onclick="removeCustomColumn('${col.key}')">删除</button>` : ''}
                            </div>
                        </div>
                    `;
            });
            container.innerHTML = html;
            document.getElementById('newColumnName').value = '';
        }

        function addCustomColumn() {
            const input = document.getElementById('newColumnName');
            const name = input.value.trim();
            if (!name) {
                showToast('请输入列名称', 'error');
                return;
            }
            const settings = getSettings();
            if (settings.some(c => c.label.toLowerCase() === name.toLowerCase())) {
                showToast('该列名已存在', 'error');
                return;
            }
            const key = 'col_' + Date.now().toString(36) + Math.random().toString(36).slice(2, 6);
            settings.push({ key, label: name, fixed: false });
            saveSettingsToStorage(settings);
            renderSettingsList();
            showToast('✅ 已添加列: ' + name, 'success');
        }

        function removeCustomColumn(key) {
            if (!confirm('确定删除该自定义列吗？删除后学生数据中的该列信息将不再显示，但数据本身不会丢失。')) return;
            let settings = getSettings();
            settings = settings.filter(c => c.key !== key);
            saveSettingsToStorage(settings);
            renderSettingsList();
            showToast('🗑️ 已删除列', 'success');
            render();
        }

        function saveSettings() {
            const settings = getSettings();
            let modified = false;
            settings.forEach(col => {
                const input = document.getElementById('setting_label_' + col.key);
                if (input) {
                    const newLabel = input.value.trim() || col.label;
                    if (newLabel !== col.label) {
                        col.label = newLabel;
                        modified = true;
                    }
                }
            });
            if (modified) {
                saveSettingsToStorage(settings);
                showToast('✅ 表头已更新', 'success');
            } else {
                showToast('没有更改', 'info');
            }
            closeSettings();
            render();
        }
